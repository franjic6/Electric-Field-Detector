void setup()
{
  Serial.begin(9600);
    pinMode(PIN_ANTENNA, INPUT);
    
    display.begin(SH1106_SWITCHCAPVCC, SCREEN_ADDRESS);
     display.clearDisplay();
  display.drawBitmap(0, 0,matrica, 128, 64, WHITE); // display.drawBitmap(x position, y position, bitmap data, bitmap width, bitmap height, color)
  display.display();
  delay(2000);
  display.clearDisplay();
  display.setTextSize(2);
  display.setTextColor(WHITE);
  display.setCursor(22,10);
  display.println("mircemk");// Print text
  display.setCursor(10,40);
  display.println("EMF METER");
  display.display();
  delay(2000);
}

void showReadings(int emfVrijednost)
{
    display.clearDisplay();   
    display.setTextSize(3);
    display.setTextColor(WHITE);
    display.setCursor(5,40);
    display.println("EMF");

    display.setCursor(70,40);
    display.println(emfVrijednost);

    display.display();
}

void loop()
{

   static int prosjecnaVrijednost = 0, emfVrijednost = 0;
    static long nextCheck = 0, emfZbroj = 0, iterations = 0;

    emfVrijednost = constrain(analogRead(PIN_ANTENNA), 0, 1023);
    emfZbroj += emfVrijednost;
    iterations++;

    if (lmillis() - nextCheck >= 0) {
        prosjecnaVrijednost = emfZbroj / iterations;
        emfZbroj = 0;
        iterations = 0;
        showReadings(prosjecnaVrijednost);
        nextCheck = lmillis() + CHECK_DELAY;
    }
    display.drawRoundRect(0, 5, 126, 30, 2, WHITE);
    display.fillRect(5, 10, 120, 23, BLACK);
    display.fillRect(5, 10, map(emfVrijednost, 0, 1023, 0, 118), 20, WHITE);
    display.display();

if (emfVrijednost > 80 && emfVrijednost <150) {
  tone (12,100,500);
  }
if (emfVrijednost > 151 && emfVrijednost <250) {
  tone (12,500,500);
}
if (emfVrijednost > 251 ) {
  tone (12,1000,500);
}
}
